extern/bolt/               # git submodule -> github.com/Ugbot/bolt
include/boltapi/
  boltapi.h                # umbrella
  version.h                # version + bolt_integration_ok()

  util/http_fuzz.h         # seeded HTTP/1.1 fuzz request generator (header-only, std-only)
